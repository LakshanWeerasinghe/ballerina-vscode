import ballerinax/snapshotdemo;

// --- Types ---

# Configuration for the demo listener.

type ListenerConfiguration record {
    # TLS material
    string secureSocket?;
    decimal timeout = 30;
};

# The record constraining @snapshotdemo:ServiceConfig.

type DemoServiceConfig record {
    # Mandatory — an empty {} would not compile
    string path;
    int retries?;
};

# Kinds of event the service can receive.
enum EventKind {
    CREATED,
    UPDATED,
    DELETED
}

# One delivered event.

type Event record {
    EventKind kind;
    http:Request payload; // Special Agent Note: Request FROM ballerina/http package
};

// --- Service ---

# Requires: import ballerinax/snapshotdemo.driver as _;
# Mandatory: this service must carry the @snapshotdemo:ServiceConfig annotation. Replace {...} with its fields, which are those of DemoServiceConfig.
@snapshotdemo:ServiceConfig {...} // required
service snapshotdemo:Service on new snapshotdemo:Listener(int port, ListenerConfiguration config = {}) {
    # Handles one event.
    # Optional parameters (may be omitted): caller
    remote function onEvent(Event event, Caller caller) returns error?;
}

# Optional: this service may carry the @othermod:OtherConfig annotation. Replace {...} with its fields, which are those of othermod:OtherServiceConfig.
@othermod:OtherConfig {...} // optional; Special Agent Note: OtherConfig, OtherServiceConfig FROM ballerinax/othermod package
service othermod:Service on new snapshotdemo:Listener() { // Special Agent Note: Service FROM ballerinax/othermod package
}

// --- Service (generic) ---
// Service Type: OpenService
// Listener: snapshotdemo:OpenListener(int port)
// Instructions:

// --- Annotations ---

# Configures the demo service.
public annotation DemoServiceConfig ServiceConfig on service;
